-- дальше внутри psql копируйте блоки:

-- список объектов
\df fn_*
SELECT tgname FROM pg_trigger t JOIN pg_class c ON c.oid=t.tgrelid
 WHERE NOT tgisinternal AND relname='grades';

-- 1) средняя по предмету+группе
SELECT fn_avg_grade_subject_group('Архитектура ЭВМ', 'ИВТ-101');

-- 2) средняя по предмету+направлению
SELECT fn_avg_grade_subject_direction(
  'Архитектура ЭВМ', 'Информатика и вычислительная техника');

-- 3) отличники
SELECT fn_count_excellent_in_group('ИВТ-101');

-- 4) несданные
SELECT id, last_name, fn_has_failed_exams(id) FROM students LIMIT 12;

-- 5–8) пропуски
SELECT * FROM fn_absences_by_student() LIMIT 20;
SELECT * FROM fn_absences_by_group();
SELECT * FROM fn_absences_by_direction();
SELECT * FROM fn_absences_by_teacher();

-- T1 (ожидается ERROR)
UPDATE grades SET grade = 7
 WHERE student_id = (SELECT student_id FROM grades LIMIT 1)
   AND subject_id = (SELECT subject_id FROM grades LIMIT 1);

-- T2 (ожидается ERROR) — чужой, но существующий преподаватель
UPDATE grades g SET set_by_teacher_id = (
  SELECT t.id FROM teachers t
   WHERE NOT EXISTS (
     SELECT 1 FROM teacher_subjects ts
      WHERE ts.teacher_id=t.id AND ts.subject_id=g.subject_id)
   LIMIT 1)
 WHERE ctid = (SELECT ctid FROM grades WHERE grade IS NOT NULL LIMIT 1);

-- T3–T5 таблицы средних
SELECT * FROM avg_group_subject LIMIT 10;
SELECT * FROM avg_group_all;
SELECT * FROM avg_subject;

-- доп. параметры
SELECT * FROM extra_param_defs;
SELECT * FROM fn_numeric_param_stats('ИВТ-101', 'часы_практики');
SELECT * FROM fn_search_text_param('Python') LIMIT 10;
SELECT fn_search_text_param_map('Python');

\q
